Convertir:
    push bp
    mov bp,sp
    sub sp,4; hacemos espacio en la pila para guardar el puntero destino
    mov efx,bp
    sub efx,4;guarda el puntero destino en efx
    mov [efx],null 
    mov edx,[bp+8]
    push efx; doble puntero al destino
    push edx; doble puntero al origen
    call move_ABB
    add sp,8
    mov [edx],[efx]; LINEA MAS IMPORTANTE!! esto cambia a donde apunta el puntero almacenado en edx(doble puntero) a lo que apunta el puntero destino(1er nodo del arbol)
Convertir_fin:

    mov sp,bp
    pop bp
    ret
