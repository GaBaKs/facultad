	MOV [4], 0
	MOV EDX, DS
	MOV ECX, 0401 	;LO MISMO QUE LDH ECX 4 Y LDL ECX 1
	MOV EAX, 1
	SYS 1
	OR EBX, [EDX] 	;COPIA EN EBX EL VALOR APUNTADO POR EDX
	SHR EBX, 1 	;DIVIDO EBX POR 2
Inicio:	CMP EBX, 1
	JZ Primo
	OR EEX, [EDX] 	;COPIA EN EEX EL VALOR APUNTADO POR EDX
	DIV EEX, EBX
	CMP AC, 0
	JZ Fin
	SUB EBX, 1
	JMP Inicio
Primo:	MOV [4], 1 	;DEVUELVO 1 PQ ES PRIMO
Fin:	MOV EAX, 1 	;COMO NO ES PRIMO [4] QUEDA EN 0
	MOV ECX, 0401 	;DEVUELVO UN UNICO VALOR DE 4 BYTES
	ADD EDX, 4 	;DEJO A EDX APUNTANDO LA CELDA [4]
	SYS 2 		;ESCRIBO EL CONTENIDO DE LA CELDA [4]
	STOP
